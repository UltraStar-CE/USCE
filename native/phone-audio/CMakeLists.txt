cmake_minimum_required(VERSION 3.24)
project(usce_phone_audio LANGUAGES C CXX)
include(FetchContent)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(NO_MEDIA ON CACHE BOOL "" FORCE)
set(NO_EXAMPLES ON CACHE BOOL "" FORCE)
set(NO_TESTS ON CACHE BOOL "" FORCE)
set(USE_GNUTLS OFF CACHE BOOL "" FORCE)
set(USE_NICE OFF CACHE BOOL "" FORCE)
set(OPENSSL_USE_STATIC_LIBS TRUE CACHE BOOL "Avoid an extra OpenSSL runtime installation")
FetchContent_Declare(datachannel GIT_REPOSITORY https://github.com/paullouisageneau/libdatachannel.git GIT_TAG 6b1e2e620f1e37f0eafeee702eaea0043cb305fd)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz
  URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa)
FetchContent_MakeAvailable(datachannel json)
add_library(usce-phone-audio SHARED phone_audio.cpp)
set_target_properties(usce-phone-audio PROPERTIES PREFIX "")
if(NOT WIN32)
  set_target_properties(usce-phone-audio PROPERTIES PREFIX "lib")
endif()
target_link_libraries(usce-phone-audio PRIVATE datachannel nlohmann_json::nlohmann_json)
target_compile_definitions(usce-phone-audio PRIVATE USCE_PHONE_AUDIO_BUILD)
if(MSVC)
  target_compile_options(usce-phone-audio PRIVATE /W4)
else()
  target_compile_options(usce-phone-audio PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(phone-audio-test phone_audio_test.cpp)
  target_link_libraries(phone-audio-test PRIVATE usce-phone-audio)
  if(MSVC)
    target_compile_options(phone-audio-test PRIVATE /UNDEBUG)
  else()
    target_compile_options(phone-audio-test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME pcm-packets COMMAND phone-audio-test)
  add_executable(relay-socket-test relay_socket_test.cpp)
  target_link_libraries(relay-socket-test PRIVATE usce-phone-audio datachannel)
  if(MSVC)
    target_compile_options(relay-socket-test PRIVATE /UNDEBUG)
  else()
    target_compile_options(relay-socket-test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME relay-socket COMMAND relay-socket-test)
  set_tests_properties(relay-socket PROPERTIES TIMEOUT 30)
endif()
