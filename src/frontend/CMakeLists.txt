cmake_minimum_required(VERSION 3.20)
project(usce_document_ui LANGUAGES C CXX)
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(RMLUI_SAMPLES OFF CACHE BOOL "" FORCE)
set(RMLUI_SVG_PLUGIN ON CACHE BOOL "" FORCE)
set(RMLUI_THIRDPARTY_CONTAINERS OFF CACHE BOOL "" FORCE)
set(LUNASVG_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(lunasvg GIT_REPOSITORY https://github.com/sammycage/lunasvg.git
  GIT_TAG 7b6970f40e72826749d4b050d11c2a4c3406f10c)
FetchContent_MakeAvailable(lunasvg)
FetchContent_Declare(rmlui GIT_REPOSITORY https://github.com/mikke89/RmlUi.git
  GIT_TAG 58c751531f4c1a1eb2dc56f409b8e8a323a0c9ad)
FetchContent_MakeAvailable(rmlui)
include(embed-effects.cmake)
include(embed-data-views.cmake)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/document-ui-licenses")
configure_file("${rmlui_SOURCE_DIR}/LICENSE.txt" "${CMAKE_CURRENT_BINARY_DIR}/document-ui-licenses/RmlUi.txt" COPYONLY)
configure_file("${lunasvg_SOURCE_DIR}/LICENSE" "${CMAKE_CURRENT_BINARY_DIR}/document-ui-licenses/LunaSVG.txt" COPYONLY)
configure_file("${lunasvg_SOURCE_DIR}/plutovg/LICENSE" "${CMAKE_CURRENT_BINARY_DIR}/document-ui-licenses/PlutoVG.txt" COPYONLY)
find_package(OpenGL REQUIRED)
find_path(SDL2_INCLUDE_DIR SDL.h PATH_SUFFIXES SDL2 REQUIRED)
find_library(SDL2_LIBRARY NAMES SDL2 REQUIRED)
find_library(SDL2_IMAGE_LIBRARY NAMES SDL2_image REQUIRED)
include(embed-renderer.cmake)
add_library(usce-document-ui SHARED document-ui.cpp document-model.cpp
  ${CMAKE_CURRENT_BINARY_DIR}/embedded-renderer.cpp)
target_compile_features(usce-document-ui PRIVATE cxx_std_17)
target_include_directories(usce-document-ui PRIVATE ${rmlui_SOURCE_DIR}/Backends ${SDL2_INCLUDE_DIR})
target_link_libraries(usce-document-ui PRIVATE RmlUi::Core lunasvg::lunasvg ${SDL2_LIBRARY} ${SDL2_IMAGE_LIBRARY} OpenGL::GL)
set_target_properties(usce-document-ui PROPERTIES PREFIX "")
option(USCE_FRONTEND_TESTS "Build app-only OpenGL integration tests" OFF)
if(USCE_FRONTEND_TESTS)
  enable_testing()
  add_executable(document-ui-test smoke-test.cpp)
  target_compile_features(document-ui-test PRIVATE cxx_std_17)
  target_include_directories(document-ui-test PRIVATE ${rmlui_SOURCE_DIR}/Backends ${SDL2_INCLUDE_DIR})
  target_link_libraries(document-ui-test PRIVATE usce-document-ui ${SDL2_LIBRARY} OpenGL::GL)
  add_test(NAME document-ui COMMAND document-ui-test "${CMAKE_CURRENT_SOURCE_DIR}/../../game")
endif()
